cmake_minimum_required(VERSION 3.8)
project(rover_perception)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(scan_filter src/scan_filter.cpp)
target_link_libraries(scan_filter rclcpp::rclcpp ${sensor_msgs_TARGETS} tf2_ros::tf2_ros)
install(TARGETS scan_filter DESTINATION lib/${PROJECT_NAME})
ament_package()
